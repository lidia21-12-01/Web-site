function updateMovement(dt){if(!st.running||!$('#taskModal').classList.contains('hidden')||!$('#infoModal').classList.contains('hidden')){animateHuman(playerRig,dt,false);return}
  let f=0,s=0;if(st.keys['w']||st.keys['arrowup'])f+=1;if(st.keys['s']||st.keys['arrowdown'])f-=1;if(st.keys['d']||st.keys['arrowright'])s+=1;if(st.keys['a']||st.keys['arrowleft'])s-=1;
  if(!f&&!s){animateHuman(playerRig,dt,false);return}
  forward.set(-Math.sin(cameraYaw),0,-Math.cos(cameraYaw)).normalize();right.set(forward.z,0,-forward.x);moveVec.set(0,0,0).addScaledVector(forward,f).addScaledVector(right,s).normalize();
  const speed=3.15;const nx=player.position.x+moveVec.x*speed*dt,nz=player.position.z+moveVec.z*speed*dt;
  if(!collisionAt(nx,player.position.z))player.position.x=nx;if(!collisionAt(player.position.x,nz))player.position.z=nz;
  const targetRot=Math.atan2(moveVec.x,moveVec.z);let diff=((targetRot-player.rotation.y+Math.PI)%(Math.PI*2))-Math.PI;player.rotation.y+=diff*Math.min(1,dt*11);animateHuman(playerRig,dt,true);
}
